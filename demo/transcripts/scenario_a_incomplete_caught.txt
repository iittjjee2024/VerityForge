     def list_products(page: int = 1, page_size: int = None):
         """Return products, optionally paginated.
         * When ``page_size`` is ``None`` (default), the function returns the full list of products.
         * When ``page_size`` is provided, it returns a dict with keys ``products`` and ``metadata`` as before.
         """
         all_items = all_products()

         # No pagination requested: return full list
         if page_size is None:

         end = start + page_size
